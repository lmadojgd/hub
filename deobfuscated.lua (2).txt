function t2.formatNumber(p4)
    if p4 >= 1E+18 then
        return string.format("%.1fqi", p4 / 1E+18)
    end

        return string.format("%.1fqa", p4 / 1000000000000000)
    end

    if p4 >= 1000000000000 then
        return string.format("%.1ft", p4 / 1000000000000)
    end

    if p4 >= 1000000000 then
        return string.format("%.1fb", p4 / 1000000000)
    end

    if p4 >= 1000000 then
        return string.format("%.1fm", p4 / 1000000)
    end

        return string.format("%.1fk", p4 / 1000)
    end

    return tostring(p4)
end

local function v203(p271)
    if p271 == nil or p271 ~= p271 then
        return "0"
    end

    local v861 = math.abs(p271)
    local v862 = p271 < 0 and "-"

    if not v862 then
        v862 = ""
    end

    if v861 >= 1E+18 then
        return v862 .. string.format("%.2fqi", v861 / 1E+18)
    end

    if v861 >= 1000000000000000 then
        return v862 .. string.format("%.2fqa", v861 / 1000000000000000)
    end

    if v861 >= 1000000000000 then
        return v862 .. string.format("%.2ft", v861 / 1000000000000)
    end

    if v861 >= 1000000000 then
        return v862 .. string.format("%.2fb", v861 / 1000000000)
    end

    if v861 >= 1000000 then
        return v862 .. string.format("%.2fm", v861 / 1000000)
    end

    if v861 >= 1000 then
        return v862 .. string.format("%.2fk", v861 / 1000)
    end

    if v861 >= 1 then
        return v862 .. string.format("%.0f", v861)
    end

    return v862 .. string.format("%.2f", v861)
end
local function v204(p272)

        return string.format("%ds", math.floor(p272))

    if p272 < 3600 then

    end

        return string.format("%dh %dm", math.floor(p272 / 3600), math.floor(p272 % 3600 / 60))
    end

    return string.format("%dd %dh", math.floor(p272 / 86400), math.floor(p272 % 86400 / 3600))
end
